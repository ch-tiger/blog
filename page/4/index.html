#!/usr/bin/env bash

# One-shot entry: smoke -> lm_eval -> msprobe, for H20 vs Ascend NPU.

#

# Run on the H20 host and on the NPU host separately, then bring the output

# files together for the diff steps. This script does NOT assume the two

# hosts can reach each other; each side produces artifacts you copy over.

#

#   bash run_all.sh <side> <model-path>

#       side: h20 | npu

#

# On the NPU side, `source /usr/local/Ascend/ascend-toolkit/set_env.sh` first

# and `pip install -e ./msprobe_dump_plugin` once.

set -euo pipefail

  

SIDE="${1:?usage: run_all.sh <h20|npu> <model-path>}"

MODEL="${2:?model path required}"

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"

DEVICE=$([[ "$SIDE" == "npu" ) && echo npu || echo cuda)

BASE_URL="http://localhost:30000"

  

echo "==================== Step A: smoke ($SIDE) ===================="

bash "${SCRIPT_DIR}/run_smoke.sh" "${SIDE}" "${MODEL}" "${DEVICE}"

# run_smoke.sh launches+generates+shutdowns; it leaves outputs_${SIDE}.jsonl

  

echo "==================== Step B: lm_eval ($SIDE) ===================="

# Re-launch the server in the background for lm_eval (Step B needs it up).

COMMON=( --model-path "${MODEL}" --port 30000 --dtype bfloat16

         --trust-remote-code --seed 42 --context-length 4096 --device "${DEVICE}" )

if [[ "${DEVICE}" == "npu" ); then

    EXTRA=( --attention-backend ascend --disable-cuda-graph --mem-fraction-static 0.8 )

    export PYTORCH_NPU_ALLOC_CONF=expandable_segments:True

    export ASCEND_LAUNCH_BLOCKING=1

else

    EXTRA=( --mem-fraction-static 0.85 )

fi

nohup python -m sglang.launch_server "${COMMON[@]}" "${EXTRA[@]}" \

    > "${SCRIPT_DIR}/server_${SIDE}_stepB.log" 2>&1 &

SRV=$!

trap 'kill $SRV 2>/dev/null || true; wait $SRV 2>/dev/null || true' EXIT

  

# Warm up with a tiny request so the first lm_eval batch isn't cold.

python "${SCRIPT_DIR}/gen_outputs.py" --base-url "${BASE_URL}" \

    --prompts "${SCRIPT_DIR}/prompts.jsonl" --output /tmp/_warmup_${SIDE}.jsonl \

    --max-new-tokens 8 || true

  

bash "${SCRIPT_DIR}/run_lmeval.sh" "${SIDE}" "${BASE_URL}"

  

echo "==================== Step C: msprobe online dump ($SIDE) ===================="

# Requires the plugin installed: pip install -e "${SCRIPT_DIR}/msprobe_dump_plugin"

DUMP_DIR="${SCRIPT_DIR}/dump_${SIDE}"

kill $SRV 2>/dev/null || true; wait $SRV 2>/dev/null || true

trap - EXIT

  

# Re-launch with plugin enabled; it hooks ModelRunner.__init__ in-scheduler.

if ! python -c "import sglang_msprobe_dump" 2>/dev/null; then

    echo "[!] sglang_msprobe_dump not installed; run: pip install -e ${SCRIPT_DIR}/msprobe_dump_plugin"

    echo "    skipping Step C on ${SIDE}"

else

    SGLANG_PLUGINS=msprobe_dump MSDUMP_DIR="${DUMP_DIR}" MSDUMP_SIDE="${SIDE}" \

        nohup python -m sglang.launch_server "${COMMON[@]}" "${EXTRA[@]}" \

        > "${SCRIPT_DIR}/server_${SIDE}_stepC.log" 2>&1 &

    SRV=$!

    trap 'kill $SRV 2>/dev/null || true; wait $SRV 2>/dev/null || true' EXIT

    # trigger one forward so hooks fire

    python "${SCRIPT_DIR}/gen_outputs.py" --base-url "${BASE_URL}" \

        --prompts "${SCRIPT_DIR}/prompts.jsonl" --output /tmp/_dumptrigger.jsonl \

        --max-new-tokens 16

    kill $SRV 2>/dev/null || true; wait $SRV 2>/dev/null || true

    trap - EXIT

    echo "[$SIDE] dump -> ${DUMP_DIR}"

fi

  

cat <<EOF

  

==================== $SIDE done =====================

Artifacts (copy these to a single machine and diff):

  Step A:  ${SCRIPT_DIR}/outputs_${SIDE}.jsonl

  Step B:  ${SCRIPT_DIR}/results_${SIDE}/

  Step C:  ${SCRIPT_DIR}/dump_${SIDE}/

  

After both sides finish, run on any host:

  python compare_outputs.py --baseline outputs_h20.jsonl --target outputs_npu.jsonl

  python lmeval_diff.py results_h20 results_npu

  bash run_msprobe_compare.sh dump_h20 dump_npu msprobe_report

EOF

"""Diff two lm_eval result directories produced by run_lmeval.sh.

  

lm_eval writes one subdirectory per task under the output path; each contains

a results.json. We aggregate by task across the two sides and report acc delta.

  

Usage:

    python lmeval_diff.py results_h20 results_npu

"""

import argparse

import json

import os

from pathlib import Path

  
  

def find_results(root: Path):

    """Return {task: acc} from an lm_eval output dir."""

    out = {}

    for p in root.rglob("results.json"):

        try:

            data = json.loads(p.read_text(encoding="utf-8"))

        except Exception as e:  # noqa: BLE001

            print(f"  ! skip {p}: {e}")

            continue

        for task, m in data.get("results", {}).items():

            acc = m.get("acc,none", m.get("acc"))

            if acc is None:

                # humaneval uses pass@1

                acc = m.get("pass@1,none", m.get("pass@1"))

            if acc is not None:

                out[task] = acc

    return out

  
  

def main():

    ap = argparse.ArgumentParser()

    ap.add_argument("h20", help="results_h20 dir")

    ap.add_argument("npu", help="results_npu dir")

    ap.add_argument("--threshold", type=float, default=0.01,

                    help="abs acc delta threshold for FAIL (default 0.01=1%)")

    args = ap.parse_args()

  

    h = find_results(Path(args.h20))

    n = find_results(Path(args.npu))

  

    print(f"\n{'task':<14}{'H20':>10}{'NPU':>10}{'diff':>10}  status")

    print("-" * 50)

    any_fail = False

    for task in sorted(set(h) | set(n)):

        a = h.get(task)

        b = n.get(task)

        if a is None or b is None:

            print(f"{task:<14}{'MISSING':>10}{'MISSING':>10}  skip")

            continue

        d = b - a

        status = "OK" if abs(d) < args.threshold else "FAIL"

        if status == "FAIL":

            any_fail = True

        print(f"{task:<14}{a:>10.4f}{b:>10.4f}{d:>+10.4f}  {status}")

  

    print("\noverall:", "PASS (precision equivalent)" if not any_fail

          else "FAIL -> run msprobe to localize")

    print("verdict threshold: |acc diff| < %.2f%%" % (args.threshold * 100))

  
  

if __name__ == "__main__":

    main()

"""SGLang plugin that dumps per-layer forward tensors for cross-hardware

precision comparison (H20 vs Ascend NPU).

  

Loaded via setuptools entry_points under group ``sglang.srt.plugins`` and

activated with ``SGLANG_PLUGINS=msprobe_dump``. Uses the official

``HookRegistry`` (see python/sglang/srt/plugins/hook_registry.py) to wrap

``ModelRunner.__init__`` with an AROUND hook. After the original __init__

returns (model loaded), the hook walks the decoder layers and registers

forward pre/post hooks on every submodule of interest. Dumped tensors go to

``$MSDUMP_DIR`` (one .pt per tensor + index.json), keyed by

``side_layer{idx:03d}_{submodule}_{input|output}`` where ``side`` is

``$MSDUMP_SIDE`` (e.g. "h20" or "npu").

  

This is an ONLINE dump: it hooks the real server path (scheduler subprocess,

real attention backend, real KV cache), so what you compare is exactly what

the server computes. No separate offline Engine is constructed, avoiding

configuration drift between the dump run and the measured run.

  

Activation:

    # package and install once (on both H20 and NPU hosts):

    pip install -e .    # from this directory

  

    # then launch the server with the plugin enabled:

    SGLANG_PLUGINS=msprobe_dump \

    MSDUMP_DIR=/data/dump_h20 \

    MSDUMP_SIDE=h20 \

    python -m sglang.launch_server --model-path ... --device cuda

  

    # on NPU:

    SGLANG_PLUGINS=msprobe_dump \

    MSDUMP_DIR=/data/dump_npu \

    MSDUMP_SIDE=npu \

    python -m sglang.launch_server --model-path ... --device npu --attention-backend ascend

"""

import json

import logging

import os

from pathlib import Path

  

from sglang.srt.plugins.hook_registry import HookRegistry, HookType

  

logger = logging.getLogger(__name__)

  

# Step3p5DecoderLayer attributes (srt/models/step3p5.py); also covers Qwen3.5

# since the submodule names are shared. Extend for finer granularity.

LAYER_SUBMODULES = (

    "self_attn",

    "moe",

    "mlp",

    "share_expert",

    "input_layernorm",

    "post_attention_layernorm",

)

  
  

def _to_cpu(x):

    import torch

    if x is None:

        return None

    if isinstance(x, (tuple, list)):

        return [_to_cpu(t) for t in x]

    if isinstance(x, dict):

        return {k: _to_cpu(v) for k, v in x.items()}

    if torch.is_tensor(x):

        return x.detach().to("cpu", dtype=torch.float32).contiguous().clone()

    return x

  
  

def _find_decoder_layers(model):

    """Locate the decoder layer list, tolerant of nesting."""

    for path in (("model", "layers"), ("model", "model", "layers")):

        cur = model

        ok = True

        for attr in path:

            cur = getattr(cur, attr, None)

            if cur is None:

                ok = False

                break

        if ok and isinstance(cur, (list, tuple)) and cur:

            return list(cur)

    # fallback: any *layers* list-like attribute

    for attr in dir(model):

        if "layer" in attr.lower():

            v = getattr(model, attr, None)

            if isinstance(v, (list, tuple)) and v:

                return list(v)

    return []

  
  

def _make_recorder(dump_dir: Path, side: str):

    """Build (pre_hook, post_hook) closures that dump to dump_dir."""

    index = {}

  

    def _record(name, tensor):

        if tensor is None:

            return

        try:

            import torch

            torch.save(tensor, str(dump_dir / f"{name}.pt"))

            index[name] = f"{name}.pt"

        except Exception:  # noqa: BLE001

            logger.exception("dump failed for %s", name)

  

    def pre_hook(submodule, layer_idx, name_prefix):

        def _h(module, args, kwargs):

            tensor = args[0] if args else None

            _record(f"{name_prefix}_input", _to_cpu(tensor))

        return _h

  

    def post_hook(submodule, layer_idx, name_prefix):

        def _h(module, args, kwargs, output):

            _record(f"{name_prefix}_output", _to_cpu(output))

        return _h

  

    return pre_hook, post_hook, index

  
  

# Module-level holder so the AROUND hook below can attach handles and write

# the index file at shutdown.

_STATE = {"handles": [], "index": {}, "dump_dir": None}

  
  

def _model_runner_init_around(original_fn, *args, **kwargs):

    """AROUND hook on ModelRunner.__init__: run original, then attach hooks."""

    result = original_fn(*args, **kwargs)

    runner = args[0] if args else None

    dump_dir = os.environ.get("MSDUMP_DIR")

    side = os.environ.get("MSDUMP_SIDE", "side")

    if not dump_dir:

        logger.warning("MSDUMP_DIR not set; msprobe_dump plugin is no-op")

        return result

    if runner is None or getattr(runner, "model", None) is None:

        logger.warning("ModelRunner.model not available after __init__; skip dump")

        return result

  

    dump_path = Path(dump_dir)

    dump_path.mkdir(parents=True, exist_ok=True)

    layers = _find_decoder_layers(runner.model)

    logger.info("msprobe_dump: side=%s layers=%d dump_dir=%s",

                side, len(layers), dump_path)

  

    pre, post, index = _make_recorder(dump_path, side)

    _STATE["index"] = index

    _STATE["dump_dir"] = dump_path

  

    for i, layer in enumerate(layers):

        for sub in LAYER_SUBMODULES:

            mod = getattr(layer, sub, None)

            if mod is None:

                continue

            name_prefix = f"{side}_layer{i:03d}_{sub}"

            _STATE["handles"].append(

                mod.register_forward_pre_hook(pre(mod, i, name_prefix), with_kwargs=True))

            _STATE["handles"].append(

                mod.register_forward_hook(post(mod, i, name_prefix), with_kwargs=True))

  

    # write index.json now so it exists even if the process is killed later.

    _write_index()

    import atexit

    atexit.register(_write_index)

    return result

  
  

def _write_index():

    dump_dir = _STATE["dump_dir"]

    if dump_dir is None:

        return

    try:

        (dump_dir / "index.json").write_text(

            json.dumps(_STATE["index"], ensure_ascii=False, indent=2),

            encoding="utf-8")

    except Exception:  # noqa: BLE001

        logger.exception("write index.json failed")

  
  

def _register():

    HookRegistry.register(

        "sglang.srt.model_executor.model_runner.ModelRunner.__init__",

        _model_runner_init_around,

        HookType.AROUND,

    )

    logger.info("msprobe_dump plugin registered AROUND ModelRunner.__init__")
