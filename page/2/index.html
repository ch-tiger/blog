"""Compare two output files produced by gen_outputs.py.

  

Reports token-level match rate and text-level similarity, then writes a JSON

report with per-sample stats and the worst-diverging prompts.

"""

import argparse

import difflib

import json

import statistics

  
  

def load(path):

    with open(path, "r", encoding="utf-8") as f:

        return [json.loads(l) for l in f if l.strip()]

  
  

def token_match_rate(ids_a, ids_b):

    n = min(len(ids_a), len(ids_b))

    if n == 0:

        return 0.0, 0

    hit = sum(1 for i in range(n) if ids_a[i] == ids_b[i])

    return hit / n, n

  
  

def text_similarity(a, b):

    return difflib.SequenceMatcher(None, a, b).ratio()

  
  

def main():

    ap = argparse.ArgumentParser()

    ap.add_argument("--baseline", required=True)

    ap.add_argument("--target", required=True)

    ap.add_argument("--out", default="compare_report.json")

    args = ap.parse_args()

  

    base = load(args.baseline)

    tgt = load(args.target)

    assert len(base) == len(tgt), f"count mismatch: {len(base)} vs {len(tgt)}"

  

    rows = []

    for a, b in zip(base, tgt):

        assert a["id"] == b["id"], f"id mismatch: {a['id']} vs {b['id']}"

        tm, n = token_match_rate(a["output_ids"], b["output_ids"])

        ts = text_similarity(a["output_text"], b["output_text"])

        rows.append({

            "id": a["id"],

            "len_base": len(a["output_ids"]),

            "len_target": len(b["output_ids"]),

            "token_match_rate": round(tm, 4),

            "text_similarity": round(ts, 4),

        })

  

    avg_tm = statistics.mean(r["token_match_rate"] for r in rows)

    avg_ts = statistics.mean(r["text_similarity"] for r in rows)

    len_base = statistics.mean(r["len_base"] for r in rows)

    len_target = statistics.mean(r["len_target"] for r in rows)

    len_ratio = len_target / len_base if len_base else 0.0

  

    print(f"\n=== Summary ({len(rows)} samples) ===")

    print(f"avg token match rate : {avg_tm:.4f}")

    print(f"avg text similarity  : {avg_ts:.4f}")

    print(f"avg len baseline     : {len_base:.1f}")

    print(f"avg len target       : {len_target:.1f}  (ratio {len_ratio:.3f})")

  

    worst = sorted(rows, key=lambda r: r["text_similarity"])[:5]

    print("\nworst 5 samples:")

    for r in worst:

        print(f"  id={r['id']:>3}  tok_match={r['token_match_rate']:.3f}  "

              f"text_sim={r['text_similarity']:.3f}  "

              f"len {r['len_base']}->{r['len_target']}")

  

    # verdict per H20-vs-NPU experience thresholds

    ok_tm = avg_tm > 0.85

    ok_ts = avg_ts > 0.92

    ok_len = 0.9 <= len_ratio <= 1.1

    print("\n=== Verdict (H20 vs NPU thresholds) ===")

    print(f"  token match > 0.85 : {ok_tm}")

    print(f"  text sim    > 0.92 : {ok_ts}")

    print(f"  len ratio in [0.9,1.1] : {ok_len}")

    print(f"  overall: {'PASS -> proceed to lm_eval' if (ok_tm and ok_ts and ok_len) else 'FAIL -> run msprobe'}")

  

    with open(args.out, "w", encoding="utf-8") as f:

        json.dump({

            "summary": {

                "avg_token_match_rate": avg_tm,

                "avg_text_similarity": avg_ts,

                "avg_len_base": len_base,

                "avg_len_target": len_target,

                "len_ratio": len_ratio,

            },

            "rows": rows,

        }, f, ensure_ascii=False, indent=2)

    print(f"\nreport written to {args.out}")

  
  

if __name__ == "__main__":

    main()

"""Fallback per-tensor comparison used by run_msprobe_compare.sh.

  

Pairs tensors by their recorded name (which encodes side_layer_idx_submodule_kind)

and reports max_abs_diff / mean_abs_diff / cosine per pair, plus the first layer

where divergence exceeds a threshold. This mirrors what msprobe produces so the

report is useful even when the official tool isn't installed.

"""

import argparse

import json

from pathlib import Path

  

import torch

  
  

def cosine(a, b):

    fa = a.flatten().float()

    fb = b.flatten().float()

    denom = fa.norm() * fb.norm()

    if denom == 0:

        return 1.0

    return (fa @ fb / denom).item()

  
  

def main():

    ap = argparse.ArgumentParser()

    ap.add_argument("--h20", required=True)

    ap.add_argument("--npu", required=True)

    ap.add_argument("--out", required=True)

    ap.add_argument("--threshold", type=float, default=0.01,

                    help="max_abs_diff above this marks a pair as DIVERGED")

    args = ap.parse_args()

  

    h_idx = json.loads((Path(args.h20) / "index.json").read_text(encoding="utf-8"))

    n_idx = json.loads((Path(args.npu) / "index.json").read_text(encoding="utf-8"))

  

    # Names differ only by side prefix; normalize by stripping "h20_"/"npu_".

    def norm(name): return name.split("_", 1)[1]

    h_map = {norm(k): k for k in h_idx}

    n_map = {norm(k): k for k in n_idx}

    common = sorted(set(h_map) & set(n_map))

  

    rows = []

    first_diverged = None

    for key in common:

        h_path = Path(args.h20) / h_idx[h_map[key)

        n_path = Path(args.npu) / n_idx[n_map[key)

        if not h_path.exists() or not n_path.exists():

            continue

        a = torch.load(str(h_path), map_location="cpu")

        b = torch.load(str(n_path), map_location="cpu")

        if torch.is_tensor(a) and torch.is_tensor(b):

            if a.shape != b.shape:

                rows.append({"name": key, "status": "SHAPE_MISMATCH",

                             "shape_a": list(a.shape), "shape_b": list(b.shape)})

                continue

            diff = (a.float() - b.float()).abs()

            mad = diff.max().item()

            mean = diff.mean().item()

            cos = cosine(a, b)

            status = "DIVERGED" if mad > args.threshold else "ok"

            if status == "DIVERGED" and first_diverged is None:

                first_diverged = key

            rows.append({"name": key, "max_abs_diff": mad,

                         "mean_abs_diff": mean, "cosine": round(cos, 6),

                         "status": status})

  

    rows.sort(key=lambda r: -r.get("max_abs_diff", 0))

    print(f"\n{'name':<40}{'max_diff':>12}{'mean_diff':>12}{'cosine':>10}  status")

    print("-" * 78)

    for r in rows[:30]:

        if r["status"] == "SHAPE_MISMATCH":

            print(f"{r['name']:<40}{'-':>12}{'-':>12}{'-':>10}  SHAPE_MISMATCH")

        else:

            print(f"{r['name']:<40}{r['max_abs_diff']:>12.6f}"

                  f"{r['mean_abs_diff']:>12.6f}{r['cosine']:>10.4f}  {r['status']}")

  

    n_div = sum(1 for r in rows if r.get("status") == "DIVERGED")

    print(f"\n{len(rows)} pairs, {n_div} DIVERGED (threshold {args.threshold})")

    if first_diverged:

        print(f"first diverged tensor: {first_diverged}")

    Path(args.out).write_text(

        json.dumps({"rows": rows, "n_diverged": n_div,

                    "first_diverged": first_diverged}, ensure_ascii=False, indent=2),

        encoding="utf-8")

    print(f"report written to {args.out}")

  
  

if __name__ == "__main__":

    main()

#!/usr/bin/env bash

# Step B: lm_eval-based acceptance comparison for H20 vs Ascend NPU.

#

# Requires a running SGLang server on each side (use run_smoke.sh without the

# auto-shutdown, or launch manually).

#

#   bash run_lmeval.sh <side> <base-url>

#       side : h20 | npu

#

# Tasks chosen for signal/noise balance on Qwen3.5-Flash:

#   gsm8k   math reasoning (sensitive to numeric precision)

#   mmlu    broad knowledge

#   humaneval  code generation (exact-match)

set -euo pipefail

  

SIDE="${1:?usage: run_lmeval.sh <h20|npu> <base-url>}"

BASE_URL="${2:?base url required, e.g. http://h20-host:30000}"

  

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"

OUT_DIR="${SCRIPT_DIR}/results_${SIDE}"

TASKS="gsm8k,mmlu,humaneval"

MODEL_NAME="${MODEL_NAME:-Qwen3.5-Flash}"

  

echo "[$SIDE] lm_eval -> ${OUT_DIR}"

lm_eval \

    --model sglang \

    --model_args "base_url=${BASE_URL},model=${MODEL_NAME}" \

    --tasks "${TASKS}" \

    --num_fewshot 0 \

    --batch_size 8 \

    --seed 42 \

    --output_path "${OUT_DIR}"

  

echo "[$SIDE] done: ${OUT_DIR}"

#

#   1) msprobe compare   — official Ascend tool, richest statistics

#   2) _torch_compare.py — pure-torch fallback (max/mean diff, cosine)

#

#   bash run_msprobe_compare.sh <dump_h20> <dump_npu> <out_dir>

mkdir -p "${OUT_DIR}"

  

if python -c "import msprobe" 2>/dev/null; then

    echo "[msprobe] running official compare"

    python -m msprobe.compare \

        --baseline "${H20_DIR}" \

        --output "${OUT_DIR}" \

fi

python "${SCRIPT_DIR}/_torch_compare.py" \

    --h20 "${H20_DIR}" --npu "${NPU_DIR}" --out "${OUT_DIR}/torch_report.json"

#!/usr/bin/env bash

# One-shot entry: smoke -> lm_eval -> msprobe, for H20 vs Ascend NPU.

#

# Run on the H20 host and on the NPU host separately, then bring the output

# files together for the diff steps. This script does NOT assume the two

# hosts can reach each other; each side produces artifacts you copy over.

#

#   bash run_all.sh <side> <model-path>

#       side: h20 | npu

#

# On the NPU side, `source /usr/local/Ascend/ascend-toolkit/set_env.sh` first

# and `pip install -e ./msprobe_dump_plugin` once.

set -euo pipefail

  

SIDE="${1:?usage: run_all.sh <h20|npu> <model-path>}"

MODEL="${2:?model path required}"

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"

DEVICE=$([[ "$SIDE" == "npu" ) && echo npu || echo cuda)

BASE_URL="http://localhost:30000"

  

echo "==================== Step A: smoke ($SIDE) ===================="

bash "${SCRIPT_DIR}/run_smoke.sh" "${SIDE}" "${MODEL}" "${DEVICE}"

# run_smoke.sh launches+generates+shutdowns; it leaves outputs_${SIDE}.jsonl

  

echo "==================== Step B: lm_eval ($SIDE) ===================="

# Re-launch the server in the background for lm_eval (Step B needs it up).

COMMON=( --model-path "${MODEL}" --port 30000 --dtype bfloat16

         --trust-remote-code --seed 42 --context-length 4096 --device "${DEVICE}" )

if [[ "${DEVICE}" == "npu" ); then

    EXTRA=( --attention-backend ascend --disable-cuda-graph --mem-fraction-static 0.8 )

    export PYTORCH_NPU_ALLOC_CONF=expandable_segments:True

    export ASCEND_LAUNCH_BLOCKING=1

else

    EXTRA=( --mem-fraction-static 0.85 )

fi

nohup python -m sglang.launch_server "${COMMON[@]}" "${EXTRA[@]}" \

    > "${SCRIPT_DIR}/server_${SIDE}_stepB.log" 2>&1 &

SRV=$!

trap 'kill $SRV 2>/dev/null || true; wait $SRV 2>/dev/null || true' EXIT

  

# Warm up with a tiny request so the first lm_eval batch isn't cold.

python "${SCRIPT_DIR}/gen_outputs.py" --base-url "${BASE_URL}" \

    --prompts "${SCRIPT_DIR}/prompts.jsonl" --output /tmp/_warmup_${SIDE}.jsonl \

    --max-new-tokens 8 || true

  

bash "${SCRIPT_DIR}/run_lmeval.sh" "${SIDE}" "${BASE_URL}"

  

echo "==================== Step C: msprobe online dump ($SIDE) ===================="

# Requires the plugin installed: pip install -e "${SCRIPT_DIR}/msprobe_dump_plugin"

DUMP_DIR="${SCRIPT_DIR}/dump_${SIDE}"

kill $SRV 2>/dev/null || true; wait $SRV 2>/dev/null || true

trap - EXIT

  

# Re-launch with plugin enabled; it hooks ModelRunner.__init__ in-scheduler.

if ! python -c "import sglang_msprobe_dump" 2>/dev/null; then

    echo "[!] sglang_msprobe_dump not installed; run: pip install -e ${SCRIPT_DIR}/msprobe_dump_plugin"

    echo "    skipping Step C on ${SIDE}"

else

    SGLANG_PLUGINS=msprobe_dump MSDUMP_DIR="${DUMP_DIR}" MSDUMP_SIDE="${SIDE}" \

        nohup python -m sglang.launch_server "${COMMON[@]}" "${EXTRA[@]}" \

        > "${SCRIPT_DIR}/server_${SIDE}_stepC.log" 2>&1 &

    SRV=$!

    trap 'kill $SRV 2>/dev/null || true; wait $SRV 2>/dev/null || true' EXIT

    # trigger one forward so hooks fire

    python "${SCRIPT_DIR}/gen_outputs.py" --base-url "${BASE_URL}" \

        --prompts "${SCRIPT_DIR}/prompts.jsonl" --output /tmp/_dumptrigger.jsonl \

        --max-new-tokens 16

    kill $SRV 2>/dev/null || true; wait $SRV 2>/dev/null || true

    trap - EXIT

    echo "[$SIDE] dump -> ${DUMP_DIR}"

fi

  

cat <<EOF

  

==================== $SIDE done =====================

Artifacts (copy these to a single machine and diff):

  Step A:  ${SCRIPT_DIR}/outputs_${SIDE}.jsonl

  Step B:  ${SCRIPT_DIR}/results_${SIDE}/

  Step C:  ${SCRIPT_DIR}/dump_${SIDE}/

  

After both sides finish, run on any host:

  python compare_outputs.py --baseline outputs_h20.jsonl --target outputs_npu.jsonl

  python lmeval_diff.py results_h20 results_npu

  bash run_msprobe_compare.sh dump_h20 dump_npu msprobe_report

EOF

"""Diff two lm_eval result directories produced by run_lmeval.sh.

  

lm_eval writes one subdirectory per task under the output path; each contains

a results.json. We aggregate by task across the two sides and report acc delta.

  

Usage:

    python lmeval_diff.py results_h20 results_npu

"""

import argparse

import json

import os

from pathlib import Path

  
  

def find_results(root: Path):

    """Return {task: acc} from an lm_eval output dir."""

    out = {}

    for p in root.rglob("results.json"):

        try:

            data = json.loads(p.read_text(encoding="utf-8"))

        except Exception as e:  # noqa: BLE001

            print(f"  ! skip {p}: {e}")

            continue

        for task, m in data.get("results", {}).items():

            acc = m.get("acc,none", m.get("acc"))

            if acc is None:

                # humaneval uses pass@1

                acc = m.get("pass@1,none", m.get("pass@1"))

            if acc is not None:

                out[task] = acc

    return out

  
  

def main():

    ap = argparse.ArgumentParser()

    ap.add_argument("h20", help="results_h20 dir")

    ap.add_argument("npu", help="results_npu dir")

    ap.add_argument("--threshold", type=float, default=0.01,

                    help="abs acc delta threshold for FAIL (default 0.01=1%)")

    args = ap.parse_args()

  

    h = find_results(Path(args.h20))

    n = find_results(Path(args.npu))

  

    print(f"\n{'task':<14}{'H20':>10}{'NPU':>10}{'diff':>10}  status")

    print("-" * 50)

    any_fail = False

    for task in sorted(set(h) | set(n)):

        a = h.get(task)

        b = n.get(task)

        if a is None or b is None:

            print(f"{task:<14}{'MISSING':>10}{'MISSING':>10}  skip")

            continue

        d = b - a

        status = "OK" if abs(d) < args.threshold else "FAIL"

        if status == "FAIL":

            any_fail = True

        print(f"{task:<14}{a:>10.4f}{b:>10.4f}{d:>+10.4f}  {status}")

  

    print("\noverall:", "PASS (precision equivalent)" if not any_fail

          else "FAIL -> run msprobe to localize")

    print("verdict threshold: |acc diff| < %.2f%%" % (args.threshold * 100))

  
  

if __name__ == "__main__":

    main()
