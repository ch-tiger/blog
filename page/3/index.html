#!/usr/bin/env bash

# Step A: smoke-test output comparison for H20 vs Ascend NPU.

#

#   bash run_smoke.sh <side> <model-path> <device>

#       side   : h20 | npu

#       device : cuda | npu

#

# On the npu side, run `source /usr/local/Ascend/ascend-toolkit/set_env.sh`

# before invoking this script.

set -euo pipefail

  

SIDE="${1:?usage: run_smoke.sh <h20|npu> <model-path> <cuda|npu>}"

MODEL="${2:?model path required}"

DEVICE="${3:?device required: cuda|npu}"

  

PORT=30000

BASE_URL="http://localhost:${PORT}"

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"

OUT="${SCRIPT_DIR}/outputs_${SIDE}.jsonl"

LOG="${SCRIPT_DIR}/server_${SIDE}.log"

  

# ---- launch server ----------------------------------------------------------

COMMON_ARGS=(

    --model-path "${MODEL}"

    --port "${PORT}"

    --dtype bfloat16

    --trust-remote-code

    --seed 42

    --context-length 4096

    --device "${DEVICE}"

)

  

if [[ "${DEVICE}" == "npu" ); then

    # Standard SGLang Ascend configuration (mirrors python/sglang/test/ascend/

    # gsm8k_ascend_mixin.py). Without --attention-backend ascend the NPU path

    # is not exercised; --disable-cuda-graph avoids graph-capture issues on NPU.

    EXTRA=(

        --attention-backend ascend

        --disable-cuda-graph

        --mem-fraction-static 0.8

    )

    export PYTORCH_NPU_ALLOC_CONF=expandable_segments:True

    export ASCEND_LAUNCH_BLOCKING=1   # sync exec for reproducibility/debug

    export TRANSFORMERS_VERBOSITY=error

else

    EXTRA=( --mem-fraction-static 0.85 )

fi

  

echo "[$SIDE] launching server: ${MODEL}"

nohup python -m sglang.launch_server "${COMMON_ARGS[@]}" "${EXTRA[@]}" \

    > "${LOG}" 2>&1 &

SERVER_PID=$!

echo "[$SIDE] server pid=${SERVER_PID}, log=${LOG}"

  

cleanup() { kill "${SERVER_PID}" 2>/dev/null || true; wait "${SERVER_PID}" 2>/dev/null || true; }

trap cleanup EXIT

  

# ---- generate outputs -------------------------------------------------------

echo "[$SIDE] generating outputs -> ${OUT}"

python "${SCRIPT_DIR}/gen_outputs.py" \

    --base-url "${BASE_URL}" \

    --prompts "${SCRIPT_DIR}/prompts.jsonl" \

    --output "${OUT}" \

    --max-new-tokens 256 \

    --seed 42 \

    --tokenizer "${MODEL}"

  

echo "[$SIDE] done: ${OUT}"

#!/usr/bin/env bash

# Step B: lm_eval-based acceptance comparison for H20 vs Ascend NPU.

#

# Requires a running SGLang server on each side (use run_smoke.sh without the

# auto-shutdown, or launch manually).

#

#   bash run_lmeval.sh <side> <base-url>

#       side : h20 | npu

#

# Tasks chosen for signal/noise balance on Qwen3.5-Flash:

#   gsm8k   math reasoning (sensitive to numeric precision)

#   mmlu    broad knowledge

#   humaneval  code generation (exact-match)

set -euo pipefail

  

SIDE="${1:?usage: run_lmeval.sh <h20|npu> <base-url>}"

BASE_URL="${2:?base url required, e.g. http://h20-host:30000}"

  

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"

OUT_DIR="${SCRIPT_DIR}/results_${SIDE}"

TASKS="gsm8k,mmlu,humaneval"

MODEL_NAME="${MODEL_NAME:-Qwen3.5-Flash}"

  

echo "[$SIDE] lm_eval -> ${OUT_DIR}"

lm_eval \

    --model sglang \

    --model_args "base_url=${BASE_URL},model=${MODEL_NAME}" \

    --tasks "${TASKS}" \

    --num_fewshot 0 \

    --batch_size 8 \

    --seed 42 \

    --output_path "${OUT_DIR}"

  

echo "[$SIDE] done: ${OUT_DIR}"

torch_npu 适配深度分析

torch_npu 通过插件机制将华为昇腾 NPU 无缝接入原生 PyTorch，其核心思路是：利用 PyTorch 预留的 PrivateUse1 后端扩展点。以下是完整适配链路：

一、架构总览

PyTorch 原生                           torch_npu 插件
┌─────────────────────┐     ┌──────────────────────────────────────┐
│  torch.Tensor       │ ──▶ │  生成 .npu() 方法                    │
│  torch.nn.Module    │ ──▶ │  生成 .npu() 方法                    │
│  torch.Storage      │ ──▶ │  生成 .npu() 方法                    │
│  DeviceType::CUDA   │     │  DeviceType::PrivateUse1 → "npu"     │
│  DispatchKey::CUDA  │ ──▶ │  DispatchKey::PrivateUse1            │
│  CUDAGuardImpl      │ ──▶ │  NPUGuardImpl (C10_REGISTER)         │
│  c10::Stream        │ ──▶ │  NPUStream                           │
│  torch._C           │ ──▶ │  torch_npu._C (pybind扩展)           │
│  torch.backends     │ ──▶ │  entry_points: torch.backends        │
│  torch.cuda.*       │ ──▶ │  torch.npu.* (镜像API)               │
│  torch.distributed  │ ──▶ │  HCCL backend (替代NCCL)             │
│  torch.compile      │ ──▶ │  NPU Inductor backend                │
│  torch.autograd     │ ──▶ │  autograd 算子注册                   │
│  CANN               │     │  (CANN/ACL底层计算库)                 │
└─────────────────────┘     └──────────────────────────────────────┘
