"""Fallback per-tensor comparison used by run_msprobe_compare.sh.

  

Pairs tensors by their recorded name (which encodes side_layer_idx_submodule_kind)

and reports max_abs_diff / mean_abs_diff / cosine per pair, plus the first layer

where divergence exceeds a threshold. This mirrors what msprobe produces so the

report is useful even when the official tool isn't installed.

"""

import argparse

import json

from pathlib import Path

  

import torch

  
  

def cosine(a, b):

    fa = a.flatten().float()

    fb = b.flatten().float()

    denom = fa.norm() * fb.norm()

    if denom == 0:

        return 1.0

    return (fa @ fb / denom).item()

  
  

def main():

    ap = argparse.ArgumentParser()

    ap.add_argument("--h20", required=True)

    ap.add_argument("--npu", required=True)

    ap.add_argument("--out", required=True)

    ap.add_argument("--threshold", type=float, default=0.01,

                    help="max_abs_diff above this marks a pair as DIVERGED")

    args = ap.parse_args()

  

    h_idx = json.loads((Path(args.h20) / "index.json").read_text(encoding="utf-8"))

    n_idx = json.loads((Path(args.npu) / "index.json").read_text(encoding="utf-8"))

  

    # Names differ only by side prefix; normalize by stripping "h20_"/"npu_".

    def norm(name): return name.split("_", 1)[1]

    h_map = {norm(k): k for k in h_idx}

    n_map = {norm(k): k for k in n_idx}

    common = sorted(set(h_map) & set(n_map))

  

    rows = []

    first_diverged = None

    for key in common:

        h_path = Path(args.h20) / h_idx[h_map[key)

        n_path = Path(args.npu) / n_idx[n_map[key)

        if not h_path.exists() or not n_path.exists():

            continue

        a = torch.load(str(h_path), map_location="cpu")

        b = torch.load(str(n_path), map_location="cpu")

        if torch.is_tensor(a) and torch.is_tensor(b):

            if a.shape != b.shape:

                rows.append({"name": key, "status": "SHAPE_MISMATCH",

                             "shape_a": list(a.shape), "shape_b": list(b.shape)})

                continue

            diff = (a.float() - b.float()).abs()

            mad = diff.max().item()

            mean = diff.mean().item()

            cos = cosine(a, b)

            status = "DIVERGED" if mad > args.threshold else "ok"

            if status == "DIVERGED" and first_diverged is None:

                first_diverged = key

            rows.append({"name": key, "max_abs_diff": mad,

                         "mean_abs_diff": mean, "cosine": round(cos, 6),

                         "status": status})

  

    rows.sort(key=lambda r: -r.get("max_abs_diff", 0))

    print(f"\n{'name':<40}{'max_diff':>12}{'mean_diff':>12}{'cosine':>10}  status")

    print("-" * 78)

    for r in rows[:30]:

        if r["status"] == "SHAPE_MISMATCH":

            print(f"{r['name']:<40}{'-':>12}{'-':>12}{'-':>10}  SHAPE_MISMATCH")

        else:

            print(f"{r['name']:<40}{r['max_abs_diff']:>12.6f}"

                  f"{r['mean_abs_diff']:>12.6f}{r['cosine']:>10.4f}  {r['status']}")

  

    n_div = sum(1 for r in rows if r.get("status") == "DIVERGED")

    print(f"\n{len(rows)} pairs, {n_div} DIVERGED (threshold {args.threshold})")

    if first_diverged:

        print(f"first diverged tensor: {first_diverged}")

    Path(args.out).write_text(

        json.dumps({"rows": rows, "n_diverged": n_div,

                    "first_diverged": first_diverged}, ensure_ascii=False, indent=2),

        encoding="utf-8")

    print(f"report written to {args.out}")

  
  

if __name__ == "__main__":

    main()
