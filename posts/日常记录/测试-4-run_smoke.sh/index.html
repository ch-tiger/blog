#!/usr/bin/env bash

# Step A: smoke-test output comparison for H20 vs Ascend NPU.

#

#   bash run_smoke.sh <side> <model-path> <device>

#       side   : h20 | npu

#       device : cuda | npu

#

# On the npu side, run `source /usr/local/Ascend/ascend-toolkit/set_env.sh`

# before invoking this script.

set -euo pipefail

  

SIDE="${1:?usage: run_smoke.sh <h20|npu> <model-path> <cuda|npu>}"

MODEL="${2:?model path required}"

DEVICE="${3:?device required: cuda|npu}"

  

PORT=30000

BASE_URL="http://localhost:${PORT}"

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"

OUT="${SCRIPT_DIR}/outputs_${SIDE}.jsonl"

LOG="${SCRIPT_DIR}/server_${SIDE}.log"

  

# ---- launch server ----------------------------------------------------------

COMMON_ARGS=(

    --model-path "${MODEL}"

    --port "${PORT}"

    --dtype bfloat16

    --trust-remote-code

    --seed 42

    --context-length 4096

    --device "${DEVICE}"

)

  

if [[ "${DEVICE}" == "npu" ); then

    # Standard SGLang Ascend configuration (mirrors python/sglang/test/ascend/

    # gsm8k_ascend_mixin.py). Without --attention-backend ascend the NPU path

    # is not exercised; --disable-cuda-graph avoids graph-capture issues on NPU.

    EXTRA=(

        --attention-backend ascend

        --disable-cuda-graph

        --mem-fraction-static 0.8

    )

    export PYTORCH_NPU_ALLOC_CONF=expandable_segments:True

    export ASCEND_LAUNCH_BLOCKING=1   # sync exec for reproducibility/debug

    export TRANSFORMERS_VERBOSITY=error

else

    EXTRA=( --mem-fraction-static 0.85 )

fi

  

echo "[$SIDE] launching server: ${MODEL}"

nohup python -m sglang.launch_server "${COMMON_ARGS[@]}" "${EXTRA[@]}" \

    > "${LOG}" 2>&1 &

SERVER_PID=$!

echo "[$SIDE] server pid=${SERVER_PID}, log=${LOG}"

  

cleanup() { kill "${SERVER_PID}" 2>/dev/null || true; wait "${SERVER_PID}" 2>/dev/null || true; }

trap cleanup EXIT

  

# ---- generate outputs -------------------------------------------------------

echo "[$SIDE] generating outputs -> ${OUT}"

python "${SCRIPT_DIR}/gen_outputs.py" \

    --base-url "${BASE_URL}" \

    --prompts "${SCRIPT_DIR}/prompts.jsonl" \

    --output "${OUT}" \

    --max-new-tokens 256 \

    --seed 42 \

    --tokenizer "${MODEL}"

  

echo "[$SIDE] done: ${OUT}"
