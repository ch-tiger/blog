#!/usr/bin/env bash

# One-shot entry: smoke -> lm_eval -> msprobe, for H20 vs Ascend NPU.

#

# Run on the H20 host and on the NPU host separately, then bring the output

# files together for the diff steps. This script does NOT assume the two

# hosts can reach each other; each side produces artifacts you copy over.

#

#   bash run_all.sh <side> <model-path>

#       side: h20 | npu

#

# On the NPU side, `source /usr/local/Ascend/ascend-toolkit/set_env.sh` first

# and `pip install -e ./msprobe_dump_plugin` once.

set -euo pipefail

  

SIDE="${1:?usage: run_all.sh <h20|npu> <model-path>}"

MODEL="${2:?model path required}"

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"

DEVICE=$([[ "$SIDE" == "npu" ) && echo npu || echo cuda)

BASE_URL="http://localhost:30000"

  

echo "==================== Step A: smoke ($SIDE) ===================="

bash "${SCRIPT_DIR}/run_smoke.sh" "${SIDE}" "${MODEL}" "${DEVICE}"

# run_smoke.sh launches+generates+shutdowns; it leaves outputs_${SIDE}.jsonl

  

echo "==================== Step B: lm_eval ($SIDE) ===================="

# Re-launch the server in the background for lm_eval (Step B needs it up).

COMMON=( --model-path "${MODEL}" --port 30000 --dtype bfloat16

         --trust-remote-code --seed 42 --context-length 4096 --device "${DEVICE}" )

if [[ "${DEVICE}" == "npu" ); then

    EXTRA=( --attention-backend ascend --disable-cuda-graph --mem-fraction-static 0.8 )

    export PYTORCH_NPU_ALLOC_CONF=expandable_segments:True

    export ASCEND_LAUNCH_BLOCKING=1

else

    EXTRA=( --mem-fraction-static 0.85 )

fi

nohup python -m sglang.launch_server "${COMMON[@]}" "${EXTRA[@]}" \

    > "${SCRIPT_DIR}/server_${SIDE}_stepB.log" 2>&1 &

SRV=$!

trap 'kill $SRV 2>/dev/null || true; wait $SRV 2>/dev/null || true' EXIT

  

# Warm up with a tiny request so the first lm_eval batch isn't cold.

python "${SCRIPT_DIR}/gen_outputs.py" --base-url "${BASE_URL}" \

    --prompts "${SCRIPT_DIR}/prompts.jsonl" --output /tmp/_warmup_${SIDE}.jsonl \

    --max-new-tokens 8 || true

  

bash "${SCRIPT_DIR}/run_lmeval.sh" "${SIDE}" "${BASE_URL}"

  

echo "==================== Step C: msprobe online dump ($SIDE) ===================="

# Requires the plugin installed: pip install -e "${SCRIPT_DIR}/msprobe_dump_plugin"

DUMP_DIR="${SCRIPT_DIR}/dump_${SIDE}"

kill $SRV 2>/dev/null || true; wait $SRV 2>/dev/null || true

trap - EXIT

  

# Re-launch with plugin enabled; it hooks ModelRunner.__init__ in-scheduler.

if ! python -c "import sglang_msprobe_dump" 2>/dev/null; then

    echo "[!] sglang_msprobe_dump not installed; run: pip install -e ${SCRIPT_DIR}/msprobe_dump_plugin"

    echo "    skipping Step C on ${SIDE}"

else

    SGLANG_PLUGINS=msprobe_dump MSDUMP_DIR="${DUMP_DIR}" MSDUMP_SIDE="${SIDE}" \

        nohup python -m sglang.launch_server "${COMMON[@]}" "${EXTRA[@]}" \

        > "${SCRIPT_DIR}/server_${SIDE}_stepC.log" 2>&1 &

    SRV=$!

    trap 'kill $SRV 2>/dev/null || true; wait $SRV 2>/dev/null || true' EXIT

    # trigger one forward so hooks fire

    python "${SCRIPT_DIR}/gen_outputs.py" --base-url "${BASE_URL}" \

        --prompts "${SCRIPT_DIR}/prompts.jsonl" --output /tmp/_dumptrigger.jsonl \

        --max-new-tokens 16

    kill $SRV 2>/dev/null || true; wait $SRV 2>/dev/null || true

    trap - EXIT

    echo "[$SIDE] dump -> ${DUMP_DIR}"

fi

  

cat <<EOF

  

==================== $SIDE done =====================

Artifacts (copy these to a single machine and diff):

  Step A:  ${SCRIPT_DIR}/outputs_${SIDE}.jsonl

  Step B:  ${SCRIPT_DIR}/results_${SIDE}/

  Step C:  ${SCRIPT_DIR}/dump_${SIDE}/

  

After both sides finish, run on any host:

  python compare_outputs.py --baseline outputs_h20.jsonl --target outputs_npu.jsonl

  python lmeval_diff.py results_h20 results_npu

  bash run_msprobe_compare.sh dump_h20 dump_npu msprobe_report

EOF
