"""Generate outputs from an SGLang server for cross-hardware precision comparison.

  

Compatible with both the native /generate endpoint and the OpenAI-compatible

/v1/completions endpoint (used by some NPU setups). Determinism is enforced via

temperature=0; pass --seed to also pin server-side RNG.

  

Usage:

    # Native SGLang endpoint (default)

    python gen_outputs.py --base-url http://localhost:30000 --output outputs.jsonl

  

    # OpenAI-compatible endpoint

    python gen_outputs.py --base-url http://npu-host:30000 --api openai \

        --model-name Qwen3.5-Flash --output outputs.jsonl

  

    # Optional: encode token ids locally if the server doesn't return them

    python gen_outputs.py --base-url ... --tokenizer /data/models/Qwen/Qwen3.5-Flash

"""

import argparse

import json

import os

import sys

import time

  

import requests

  
  

def generate_native(base_url: str, text: str, max_new_tokens: int, seed: int):

    """Call SGLang native /generate endpoint."""

    resp = requests.post(

        f"{base_url}/generate",

        json={

            "text": text,

            "sampling_params": {

                "max_new_tokens": max_new_tokens,

                "temperature": 0,   # greedy, deterministic

                "top_p": 1.0,

                "top_k": -1,

            },

            "stream": False,

        },

        timeout=300,

    )

    resp.raise_for_status()

    out = resp.json()

    return out.get("text", ""), out.get("output_ids", [])

  
  

def generate_openai(base_url: str, text: str, model_name: str,

                    max_new_tokens: int, seed: int):

    """Call OpenAI-compatible /v1/completions endpoint."""

    resp = requests.post(

        f"{base_url}/v1/completions",

        json={

            "model": model_name,

            "prompt": text,

            "max_tokens": max_new_tokens,

            "temperature": 0,

            "top_p": 1.0,

            "seed": seed,

        },

        timeout=300,

    )

    resp.raise_for_status()

    out = resp.json()

    choice = out["choices"][0]

    text_out = choice.get("text", "")

    # Try to get token ids from logprobs; fall back later if absent.

    ids = []

    logprobs = choice.get("logprobs")

    if logprobs and "token_ids" in logprobs:

        ids = list(logprobs["token_ids"])

    return text_out, ids

  
  

def wait_for_server(base_url: str, timeout: int = 600):

    """Block until the server /health responds."""

    deadline = time.time() + timeout

    last_err = None

    while time.time() < deadline:

        try:

            r = requests.get(f"{base_url}/health", timeout=5)

            if r.status_code == 200:

                return

        except Exception as e:  # noqa: BLE001

            last_err = e

        time.sleep(5)

    raise RuntimeError(f"server at {base_url} not ready within {timeout}s: {last_err}")

  
  

def main():

    ap = argparse.ArgumentParser()

    ap.add_argument("--base-url", default="http://localhost:30000")

    ap.add_argument("--api", choices=["native", "openai"], default="native")

    ap.add_argument("--model-name", default="default",

                    help="model name for OpenAI-compatible API")

    ap.add_argument("--prompts", default="prompts.jsonl")

    ap.add_argument("--output", required=True)

    ap.add_argument("--max-new-tokens", type=int, default=256)

    ap.add_argument("--seed", type=int, default=42)

    ap.add_argument("--tokenizer", default=None,

                    help="HF model path; if set, re-encode output text to get "

                         "token ids when the server omits them")

    ap.add_argument("--no-wait", action="store_true",

                    help="skip server health check")

    args = ap.parse_args()

  

    if not args.no_wait:

        print(f"waiting for server at {args.base_url} ...", flush=True)

        wait_for_server(args.base_url)

  

    tokenizer = None

    if args.tokenizer:

        from transformers import AutoTokenizer

        tokenizer = AutoTokenizer.from_pretrained(

            args.tokenizer, trust_remote_code=True)

  

    with open(args.prompts, "r", encoding="utf-8") as f:

        prompts = [json.loads(line) for line in f if line.strip()]

  

    results = []

    for i, p in enumerate(prompts):

        if args.api == "native":

            text_out, ids = generate_native(

                args.base_url, p["text"], args.max_new_tokens, args.seed)

        else:

            text_out, ids = generate_openai(

                args.base_url, p["text"], args.model_name,

                args.max_new_tokens, args.seed)

  

        if not ids and tokenizer is not None:

            ids = tokenizer.encode(text_out, add_special_tokens=False)

  

        results.append({

            "id": i,

            "prompt": p["text"],

            "output_text": text_out,

            "output_ids": list(ids),

        })

        print(f"[{i+1}/{len(prompts)}] tokens={len(ids)}", flush=True)

  

    os.makedirs(os.path.dirname(os.path.abspath(args.output)), exist_ok=True)

    with open(args.output, "w", encoding="utf-8") as f:

        for r in results:

            f.write(json.dumps(r, ensure_ascii=False) + "\n")

    print(f"saved {len(results)} outputs to {args.output}", flush=True)

  
  

if __name__ == "__main__":

    sys.exit(main())
