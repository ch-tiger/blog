核心一句话

torch._C._register_privateuse1_backend("zpu") 不需要修改 PyTorch 源码。 它只是让 PyTorch 知道 “zpu” 等于 PrivateUse1，然后你的 TORCH_LIBRARY_IMPL(sgl_kernel, PrivateUse1, m) 在 common_ops.so 加载时自然生效。你的 rmsnorm_zpu() 函数内部直接调用 zpu_ops 的 native C API，不需要依赖 PyTorch 的基础算子栈。

/* Copyright 2025 SGLang Team. All Rights Reserved. */

#include <torch/all.h>
#include <torch/library.h>

#include "sgl_kernel_ops.h"

// ============================================================
// zpu ops 声明 — 这些函数由你的 zpu_ops 库提供
// ============================================================
// 假设 zpu_ops 库提供如下 API：
// namespace zpu::op {
//   void rmsnorm(void* output, const void* input, const void* weight,
//                int batch_size, int hidden_size, float eps);
//   void fused_add_rmsnorm(void* input, void* residual, const void* weight,
//                          int batch_size, int hidden_size, float eps);
//   void gemma_rmsnorm(void* output, const void* input, const void* weight,
//                      int batch_size, int hidden_size, float eps);
//   void gemma_fused_add_rmsnorm(void* input, void* residual, const void* weight,
//                                int batch_size, int hidden_size, float eps);
// }
// 请根据你的 zpu_ops 实际头文件调整 include
#include <zpu_ops/rmsnorm.h>  // ← 替换为实际路径

// ============================================================
// rmsnorm — zpu 实现
// ============================================================
void rmsnorm_zpu(
    at::Tensor& output,
    at::Tensor& input,
    at::Tensor& weight,
    double eps,
    bool /*enable_pdl*/) {
  // 参数校验
  TORCH_CHECK(input.is_contiguous(), "rmsnorm_zpu: input must be contiguous");
  TORCH_CHECK(weight.is_contiguous(), "rmsnorm_zpu: weight must be contiguous");
  TORCH_CHECK(output.is_contiguous(), "rmsnorm_zpu: output must be contiguous");
  TORCH_CHECK(input.dim() == 2, "rmsnorm_zpu: input must be 2D");
  TORCH_CHECK(weight.dim() == 1, "rmsnorm_zpu: weight must be 1D");
  TORCH_CHECK(input.size(1) == weight.size(0),
              "rmsnorm_zpu: input last dim must match weight size");

  int batch_size = static_cast<int>(input.size(0));
  int hidden_size = static_cast<int>(input.size(1));
  float eps_f = static_cast<float>(eps);

  // 调用 zpu_ops 算子库
  zpu::op::rmsnorm(
      output.data_ptr(),
      input.data_ptr(),
      weight.data_ptr(),
      batch_size,
      hidden_size,
      eps_f);
}

// ============================================================
// fused_add_rmsnorm — zpu 实现
// ============================================================
void sgl_fused_add_rmsnorm_zpu(
    torch::Tensor input,
    torch::Tensor residual,
    torch::Tensor weight,
    double eps,
    bool /*enable_pdl*/) {
  TORCH_CHECK(input.is_contiguous(), "fused_add_rmsnorm_zpu: input must be contiguous");
  TORCH_CHECK(residual.is_contiguous(), "fused_add_rmsnorm_zpu: residual must be contiguous");
  TORCH_CHECK(weight.is_contiguous(), "fused_add_rmsnorm_zpu: weight must be contiguous");
  TORCH_CHECK(input.dim() == 2, "fused_add_rmsnorm_zpu: input must be 2D");
  TORCH_CHECK(residual.dim() == 2, "fused_add_rmsnorm_zpu: residual must be 2D");
  TORCH_CHECK(weight.dim() == 1, "fused_add_rmsnorm_zpu: weight must be 1D");
  TORCH_CHECK(input.sizes() == residual.sizes(),
              "fused_add_rmsnorm_zpu: input and residual must have same shape");
  TORCH_CHECK(input.size(1) == weight.size(0),
              "fused_add_rmsnorm_zpu: input last dim must match weight size");

  int batch_size = static_cast<int>(input.size(0));
  int hidden_size = static_cast<int>(input.size(1));
  float eps_f = static_cast<float>(eps);

  // 调用 zpu_ops 算子库
  zpu::op::fused_add_rmsnorm(
      input.data_ptr(),
      residual.data_ptr(),
      weight.data_ptr(),
      batch_size,
      hidden_size,
      eps_f);
}

// ============================================================
// gemma_rmsnorm — zpu 实现
// gemma 变体: out = (1 + weight) * rmsnorm(input)

  TORCH_CHECK(input.dim() == 2, "gemma_rmsnorm_zpu: input must be 2D");
  TORCH_CHECK(weight.dim() == 1, "gemma_rmsnorm_zpu: weight must be 1D");
  TORCH_CHECK(input.size(1) == weight.size(0),

  int batch_size = static_cast<int>(input.size(0));
  int hidden_size = static_cast<int>(input.size(1));

      input.data_ptr(),
      weight.data_ptr(),
      batch_size,
      hidden_size,
      eps_f);

// ============================================================
// gemma_fused_add_rmsnorm — zpu 实现

  TORCH_CHECK(input.dim() == 2, "gemma_fused_add_rmsnorm_zpu: input must be 2D");
  TORCH_CHECK(residual.dim() == 2, "gemma_fused_add_rmsnorm_zpu: residual must be 2D");
  TORCH_CHECK(weight.dim() == 1, "gemma_fused_add_rmsnorm_zpu: weight must be 1D");

  TORCH_CHECK(input.size(1) == weight.size(0),

  int batch_size = static_cast<int>(input.size(0));
  int hidden_size = static_cast<int>(input.size(1));
  float eps_f = static_cast<float>(eps);

  zpu::op::gemma_fused_add_rmsnorm(

// ============================================================
// ★ 关键: 注册到 PrivateUse1 dispatch key
// TORCH_LIBRARY_IMPL 只绑定实现，不重新定义 schema
// schema 已在 common_extension.cc 的 TORCH_LIBRARY_FRAGMENT 中定义
// ============================================================
TORCH_LIBRARY_IMPL(sgl_kernel, PrivateUse1, m) {
  m.impl("rmsnorm",              TORCH_FN(rmsnorm_zpu));
  m.impl("fused_add_rmsnorm",    TORCH_FN(sgl_fused_add_rmsnorm_zpu));
  m.impl("gemma_rmsnorm",        TORCH_FN(gemma_rmsnorm_zpu));
  m.impl("gemma_fused_add_rmsnorm", TORCH_FN(gemma_fused_add_rmsnorm_zpu));
}

核心原理：

  • TORCH_LIBRARY_FRAGMENT(sgl_kernel, m) 中的 m.def(...) 定义了算子的 schema（签名），这部分在 common_extension.cc 中 已经存在
  • TORCH_LIBRARY_IMPL(sgl_kernel, PrivateUse1, m) 中的 m.impl(...) 将具体实现绑定到 PrivateUse1 设备，这部分是新的
  • 当 PyTorch dispatcher 看到 device="zpu" 的 tensor 时，它自动寻找 PrivateUse1 下的注册实现

文件 5: 修改 python/sglang/srt/utils/common.py — 添加 is_zpu()

# python/sglang/srt/utils/common.py

@lru_cache(maxsize=1)
def is_zpu() -> bool:
    try:
        return torch._C._get_privateuse1_backend_name() == "zpu"
    except (AttributeError, RuntimeError):
        return False

在 RMSNorm 类中新增方法：

class RMSNorm(MultiPlatformOp):
    # ... 现有代码 ...

    def forward_zpu(
        self,
        x: torch.Tensor,
        residual: Optional[torch.Tensor] = None,
        post_residual_addition: Optional[torch.Tensor] = None,
    ) -> Union[torch.Tensor, Tuple[torch.Tensor, torch.Tensor):
        """zpu 设备的 RMSNorm 前向实现。"""
        if x.numel() == 0:
            if residual is not None:
                if post_residual_addition is not None:
                    residual = residual + post_residual_addition
                return x, residual
            return x

        # sgl_kernel rmsnorm 要求 2D 输入
        needs_reshape = x.dim() != 2 and residual is None
        if needs_reshape:
            original_shape = x.shape
            x = x.contiguous().reshape(-1, original_shape[-1])

        if self.variance_size_override is not None:
            return self.forward_native(x, residual, post_residual_addition)

        # cast_x_before_out_mul: HF 语义 — fallback 到 native
        if self.cast_x_before_out_mul and residual is None:
            out = self.forward_native(x, None, None)
            if needs_reshape:
                out = out.reshape(original_shape)
            return out

        if residual is not None:
            if post_residual_addition is not None:
                residual = residual + post_residual_addition
            fused_add_rmsnorm(x, residual, self.weight.data, self.variance_epsilon)
            return x, residual

        out = rmsnorm(x, self.weight.data, self.variance_epsilon)
        if needs_reshape:
            out = out.reshape(original_shape)
        return out

最重要的一句话

你不需要修改 PyTorch 源码。 只需要在 sgl-kernel/python/sgl_kernel/__init__.py 中调用 torch._C._register_privateuse1_backend("zpu")，然后在 zpu_extension.cc 中用 TORCH_LIBRARY_IMPL(sgl_kernel, PrivateUse1, m) 注册实现即可。PyTorch dispatcher 会自动路由。

你可以直接复制上面的代码，然后把 #include <zpu_ops/rmsnorm.h> 替换为你的 zpu_ops 库实际头文件，以及确认 zpu::op::rmsnorm 的函数签名是否匹配。需要我帮你调整什么具体细节吗
